  import math

# Coeficientes
a = float(input("Digite o valor de a: "))
b = float(input("Digite o valor de b: "))
c = float(input("Digite o valor de c: "))

# Calculando o discriminante (delta)
delta = b**2 - 4*a*c

if a == 0:
    print("Isso não é uma equação de segundo grau.")
else:
    if delta > 0:
        x1 = (-b + math.sqrt(delta)) / (2*a)
        x2 = (-b - math.sqrt(delta)) / (2*a)
        print(f"Duas raízes reais: x1 = {x1} e x2 = {x2}")
    elif delta == 0:
        x = -b / (2*a)
        print(f"Uma raiz real: x = {x}")
    else:
        real = -b / (2*a)
        imag = math.sqrt(-delta) / (2*a)
        print(f"Raízes complexas: {real} + {imag}i e {real} - {imag}i")
